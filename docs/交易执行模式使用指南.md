//! # 交易执行模式使用指南
//!
//! 本文档展示如何使用 `ExecutionMode` 在不同场景下选择最适合的交易执行方式。

use sol_trade_sdk::{
    common::SolanaRpcClient,
    trading::ExecutionMode,
    DexType, TradeBuyParams, TradeTokenType, TradeConfig,
};
use solana_sdk::{pubkey::Pubkey, signature::Keypair, signer::Signer};
use std::str::FromStr;
use std::sync::Arc;

/// 示例 1: 快速价格估算（本地计算）
///
/// **场景**: 用户在 UI 中查看"如果我用 0.1 SOL 买 JUP，能得到多少？"
/// **需求**: 快速响应，无需等待网络
/// **方案**: 使用 `ExecutionMode::LocalCalculation`
///
/// ```ignore
/// async fn example_price_estimation() -> Result<u64, Box<dyn std::error::Error>> {
///     let rpc_url = "http://127.0.0.1:8899";
///     let rpc = Arc::new(SolanaRpcClient::new(rpc_url.to_string()));
///     let payer = Arc::new(Keypair::new());
///     let config = TradeConfig::new(rpc_url.to_string(), HashMap::new(), CommitmentConfig::confirmed());
///
///     let client = TradingClient::new(payer, config).await?;
///
///     // 本地计算 - 快速返回（< 10ms）
///     let result = client.buy(TradeBuyParams {
///         dex_type: DexType::RaydiumClmm,
///         input_token_type: TradeTokenType::SOL,
///         mint: jup_mint,
///         input_token_amount: 100_000_000, // 0.1 SOL
///         slippage_basis_points: Some(100), // 1%
///         ..Default::default()
///     }).await?;
///
///     println!("预估获得: {} JUP", result.amount_out);
///     Ok(result.amount_out)
/// }
/// ```

/// 示例 2: 交易前验证（链上模拟）
///
/// **场景**: 用户点击"确认交易"前，验证交易是否成功
/// **需求**: 准确验证，但不消耗真实费用
/// **方案**: 使用 `ExecutionMode::Simulation`
///
/// ```ignore
/// async fn example_pre_trade_validation() -> Result<bool, Box<dyn std::error::Error>> {
///     let rpc_url = "http://127.0.0.1:8899";
///     let rpc = Arc::new(SolanaRpcClient::new(rpc_url.to_string()));
///     let payer = Arc::new(Keypair::new());
///     let config = TradeConfig::new(rpc_url.to_string(), HashMap::new(), CommitmentConfig::confirmed());
///
///     let client = TradingClient::new(payer, config).await?;
///
///     // 链上模拟 - 验证交易（约 1-2 秒）
///     let result = client.buy(TradeBuyParams {
///         dex_type: DexType::RaydiumClmm,
///         input_token_type: TradeTokenType::SOL,
///         mint: jup_mint,
///         input_token_amount: 100_000_000, // 0.1 SOL
///         slippage_basis_points: Some(100), // 1%
///         simulate: true, // 开启模拟模式
///         ..Default::default()
///     }).await?;
///
///     if result.success {
///         println!("✅ 模拟成功，预计获得: {} JUP", result.amount_out);
///         println!("   CU 消耗: {:?}", result.compute_units);
///         Ok(true)
///     } else {
///         println!("❌ 模拟失败: {:?}", result.error);
///         Ok(false)
///     }
/// }
/// ```

/// 示例 3: 真实交易执行
///
/// **场景**: 用户确认交易，执行真实交易
/// **需求**: 实际执行，改变链上状态
/// **方案**: 使用 `ExecutionMode::RealExecution`（默认）
///
/// ```ignore
/// async fn example_real_execution() -> Result<Signature, Box<dyn std::error::Error>> {
///     let rpc_url = "http://127.0.0.1:8899";
///     let rpc = Arc::new(SolanaRpcClient::new(rpc_url.to_string()));
///     let payer = Arc::new(Keypair::new());
///     let config = TradeConfig::new(rpc_url.to_string(), HashMap::new(), CommitmentConfig::confirmed());
///
///     let client = TradingClient::new(payer, config).await?;
///
///     // 真实执行 - 实际交易
///     let result = client.buy(TradeBuyParams {
///         dex_type: DexType::RaydiumClmm,
///         input_token_type: TradeTokenType::SOL,
///         mint: jup_mint,
///         input_token_amount: 100_000_000, // 0.1 SOL
///         slippage_basis_points: Some(100), // 1%
///         simulate: false, // 关闭模拟，真实执行
///         wait_transaction_confirmed: true,
///         ..Default::default()
///     }).await?;
///
///     if let Some(signature) = result.signatures.first() {
///         println!("✅ 交易成功，签名: {}", signature);
///         Ok(*signature)
///     } else {
///         Err("交易未返回签名".into())
///     }
/// }
/// ```

/// 示例 4: 三阶段交易流程（最佳实践）
///
/// **场景**: 完整的交易流程，包含价格估算 → 模拟验证 → 真实执行
/// **需求**: 最佳用户体验
/// **方案**: 依次使用三种模式
///
/// ```ignore
/// async fn example_complete_trading_flow() -> Result<(), Box<dyn std::error::Error>> {
///     let rpc_url = "http://127.0.0.1:8899";
///     let rpc = Arc::new(SolanaRpcClient::new(rpc_url.to_string()));
///     let payer = Arc::new(Keypair::new());
///     let config = TradeConfig::new(rpc_url.to_string(), HashMap::new(), CommitmentConfig::confirmed());
///
///     let client = TradingClient::new(payer, config).await?;
///     let wsol_mint = Pubkey::from_str("So11111111111111111111111111111111111111112")?;
///     let jup_mint = Pubkey::from_str("JUPyiwrYJFskUPiHa7hkeR8VUtAeFoSYbKedZNsDvCN")?;
///
///     // ========================================
/// // 阶段 1: 快速价格估算（本地计算）
///     // ========================================
///     println!("🔍 阶段 1: 价格估算...");
///
///     // 使用 quote_exact_in 快速计算（未来功能）
///     let estimated_output = 61_395_891u64; // 假设计算结果
///
///     println!("   输入: 0.1 SOL");
///     println!("   预估输出: {} JUP", estimated_output);
///     println!("   ⏱️  响应时间: < 10ms\n");
///
///     // ========================================
///     // 阶段 2: 链上模拟验证
///     // ========================================
///     println!("🧪 阶段 2: 模拟验证...");
///
///     let sim_result = client.buy(TradeBuyParams {
///         dex_type: DexType::RaydiumClmm,
///         input_token_type: TradeTokenType::SOL,
///         mint: jup_mint,
///         input_token_amount: 100_000_000,
///         slippage_basis_points: Some(100),
///         simulate: true,
///         wait_transaction_confirmed: false,
///         ..Default::default()
///     }).await?;
///
///     if !sim_result.success {
///         println!("❌ 模拟失败，交易可能失败");
///         return Err("模拟验证失败".into());
///     }
///
///     let simulated_output = 61_399_000u64; // 从模拟结果获取
///     let error_rate = ((simulated_output as i128 - estimated_output as i128).abs() as f64
///         / simulated_output as f64) * 100.0;
///
///     println!("   模拟输出: {} JUP", simulated_output);
///     println!("   计算误差: {:.4}%", error_rate);
///     println!("   ⏱️  响应时间: ~1-2s\n");
///
///     if error_rate > 1.0 {
///         println!("⚠️  警告：计算误差较大（> 1%），建议调整滑点");
///     }
///
///     // ========================================
///     // 阶段 3: 真实执行
///     // ========================================
///     println!("💎 阶段 3: 真实执行...");
///
///     let real_result = client.buy(TradeBuyParams {
///         dex_type: DexType::RaydiumClmm,
///         input_token_type: TradeTokenType::SOL,
///         mint: jup_mint,
///         input_token_amount: 100_000_000,
///         slippage_basis_points: Some(100),
///         simulate: false,
///         wait_transaction_confirmed: true,
///         ..Default::default()
///     }).await?;
///
///     if let Some(signature) = real_result.signatures.first() {
///         println!("✅ 交易成功！");
///         println!("   签名: {}", signature);
///         println!("   ⏱️  确认时间: ~2-5s");
///     }
///
///     Ok(())
/// }
/// ```

/// 示例 5: 批量价格查询（使用本地计算）
///
/// **场景**: 需要查询多个 DEX 的价格
/// **需求**: 快速响应，并发查询
/// **方案**: 全部使用本地计算
///
/// ```ignore
/// async fn example_batch_price_query() -> Result<Vec<(DexType, u64)>, Box<dyn std::error::Error>> {
///     let rpc = Arc::new(SolanaRpcClient::new("http://127.0.0.1:8899".to_string()));
///     let wsol_mint = Pubkey::from_str("So11111111111111111111111111111111111111112")?;
///     let jup_mint = Pubkey::from_str("JUPyiwrYJFskUPiHa7hkeR8VUtAeFoSYbKedZNsDvCN")?;
///
///     let dex_types = vec![
///         DexType::RaydiumClmm,
///         DexType::RaydiumCpmm,
///         DexType::RaydiumAmmV4,
///     ];
///
///     // 并发查询所有 DEX（使用本地计算，非常快速）
///     let mut handles = vec![];
///     for dex_type in dex_types {
///         let rpc_clone = rpc.clone();
///         handles.push(tokio::spawn(async move {
///             // 这里应该调用 quote_exact_in
///             // 暂时返回模拟数据
///             (dex_type, 61_395_891u64)
///         }));
///     }
///
///     let mut results = vec![];
///     for handle in handles {
///         let (dex_type, amount) = handle.await?;
///         results.push((dex_type, amount));
///     }
///
///     // 显示结果
///     for (dex_type, amount) in &results {
///         println!("{:?}: {} JUP", dex_type, amount);
///     }
///
///     Ok(results)
/// }
/// ```

/// 选择执行模式的决策树
///
/// ```
/// 开始
///   │
///   ├─ 需要快速响应？
///   │  ├─ 是 → 使用 LocalCalculation
///   │  │   - 价格估算
///   │  │   - UI 预览
///   │  │   - 批量查询
///   │  │
///   └─ 否 → 需要验证准确性？
///      ├─ 是 → 使用 Simulation
///      │   - 交易前验证
///      │   - 测试交易
///      │   - 计算验证
///      │
///      └─ 否 → 使用 RealExecution
///          - 正式交易
///          - 改变状态
///          - 实际获利
/// ```

/// 使用建议总结
///
/// | 模式 | 速度 | 准确性 | 费用 | 适用场景 |
/// |------|------|--------|------|---------|
/// | LocalCalculation | ⚡️ 极快 (< 10ms) | ⚠️ 估算 (±1%) | 🆓 无 | 价格查询、UI 显示 |
/// | Simulation | 🐢 较快 (1-2s) | ✅ 准确 (链上逻辑) | 🆓 无 | 交易验证、测试 |
/// | RealExecution | 🐌 较慢 (2-5s) | ✅ 实际 | 💸 有 | 正式交易 |
